# Hindi Reel Studio — direct Python dependencies.
#
# Used by HindiReelStudio.command to rebuild .venv when the environment is
# broken. Versions are intentionally unpinned: pip resolves a working tree
# and the launcher re-validates afterwards. Transitive deps (numpy, pandas,
# protobuf, …) come in automatically via streamlit.
streamlit        # UI framework (app.py)
pydantic         # data models (core/models.py)
httpx            # news fetching (tools/news_fetcher.py)
feedparser       # RSS parsing (tools/news_fetcher.py)
beautifulsoup4   # HTML cleaning (tools/news_fetcher.py)
